cmake_minimum_required(VERSION 3.16)
project(Direct3D9ScreenCapture LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "Direct3D 9 and WIC require Windows.")
endif()
add_library(png_writer png_writer.cpp)
target_include_directories(png_writer PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(png_writer PUBLIC cxx_std_17)
target_compile_definitions(png_writer PUBLIC _WIN32_WINNT=0x0A00)
target_link_libraries(png_writer PUBLIC windowscodecs ole32 uuid)
add_executable(directx-capture main.cpp)
target_link_libraries(directx-capture PRIVATE png_writer d3d9 user32)
if(MINGW)
    target_link_options(directx-capture PRIVATE -municode)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(png-test tests/png_test.cpp)
    target_link_libraries(png-test PRIVATE png_writer)
    add_test(NAME synthetic-png-roundtrip COMMAND png-test)
endif()
foreach(target png_writer directx-capture png-test)
    if(TARGET ${target})
        set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /permissive-)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    endif()
endforeach()
